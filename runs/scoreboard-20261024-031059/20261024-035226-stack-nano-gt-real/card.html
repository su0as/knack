+    for attempt in range(MAX_ATTEMPTS):

+        # 1. Perceive -------------------------------------------------------
+        obs = robot.observe()
+        objs = obs.get("objects", {})
+        red = objs.get("red_cube")
+        blue = objs.get("blue_cube")
+        if red is None or blue is None:
+            robot.log("Target cubes not found")
+            return False
+
+        # ------------------------------------------------------------------
+        # 2. Pick the red cube ---------------------------------------------
+        #   Move just above the red cube's centre (a small lift to avoid
+        #   colliding with the table)
+        rx, ry, rz = red["pos"]
+        pick_z = rz + 0.02  # 2 cm above the centre
+        try:
+            robot.move_to(rx, ry, pick_z, speed=0.07)
+        except Exception:  # catches MotionError and any unexpected exception
+            robot.log("Pick move failed")
+            return False
+
+        robot.grasp()  # close fingers around the red cube
+        if robot.holding() != "red_cube":
+            robot.log("Grasp verification failed")
+            if attempt == MAX_ATTEMPTS - 1:
+                return False
+            continue  # retry from the start
+
+        # ------------------------------------------------------------------
+        # 3. Lift and position over the blue cube -------------------------
+        robot.move_by(dz=0.02, speed=0.07)  # raise a little to clear the red cube
+
+        #   Compute the centre height where the red cube should rest on the blue cube
+        #   blue centre z + half its height + half red height = blue_z + 0.015 + 0.015
+        bx, by, bz = blue["pos"]
+        place_z = bz + 0.015 + 0.015  # 0.03 m above the table
+
+        try:
+            robot.move_to(bx, by, place_z, speed=0.07)
+        except Exception:
+            robot.log("Place move failed")
+            return False
+
+        # ------------------------------------------------------------------
+        # 4. Release --------------------------------------------------------
+        robot.release()
+
+        #   Verify that nothing is being held any more
+        if robot.holding() is None:
+            return True  # success
+        else:
+            robot.log("Release verification failed, still holding")
+            if attempt == MAX_ATTEMPTS - 1:
+                return False
+            # Attempt a simple recovery: lower back, re‑grasp and try again
+            robot.move_by(dz=-0.02, speed=0.07)

+    return False

failure analysis (summariser)
**Mode 1: Persistent grasp failure (55/56 episodes)**  
*Root cause:* The verification check at line 66 (`robot.holding() != "red_cube"`) triggers a re‑grasp after every successful `move_to`. The re‑grasp sequence (lines 68‑72) releases the cube, lifts, and re‑grasps, but the gripper often ends holding `None` because the initial `grasp()` at line 62 does not secure the cube reliably. The subsequent `move_by(dz=-0.01)` at line 72 may place the gripper out of the cube’s reach, causing the second `grasp()` to fail.  
*Fix:* Remove the automatic re‑grasp block (lines 66‑75). After the initial `grasp()` at line 62, assume success unless `holding()` returns something other than `"red_cube"`. If verification fails, log and abort the current overall attempt rather than looping a fragile re‑grasp.

**Mode 2: Placement target drift (1/56 episodes)**  
*Root cause:* During placement recovery (lines 96‑114), the code shifts the target toward the red cube by a fixed `factor=0.5` (line 112) when the offset exceeds 0.015 m. This large move can push the target past the blue cube or into an unreachable workspace, causing subsequent `move_to` failures (line 117) and aborting the episode.  
*Fix:* Reduce the adjustment factor to `0.1` and add a maximum shift limit (e.g., `max_shift=0.02`) to keep the target within a safe radius of the blue cube centre.

import math, numpy as np, itertools, functools, collections

def run(robot):

    Stack the red cube on top of the blue cube and release it.
    The implementation now includes:
      • perception sanity check with one retry,
      • seated grasp verification,
      • placement‑offset validation before release,
      • limited retries for grasp and placement,
      • simple recovery on release failure.
    """
    MAX_OVERALL = 3  # overall attempts before giving up

    for overall_attempt in range(MAX_OVERALL):
        # --------------------------------------------------------------
        # 1. Perceive (with one extra observation if needed)
        obs = robot.observe()
        objs = obs.get("objects", {})
        red = objs.get("red_cube")
        blue = objs.get("blue_cube")
        if red is None or blue is None:
            robot.log("Target cubes missing – retrying perception")
            obs2 = robot.observe()
            objs2 = obs2.get("objects", {})
            red = objs2.get("red_cube")
            blue = objs2.get("blue_cube")
            if red is None or blue is None:
                robot.log("Cubes still missing after retry")
                return False

        # --------------------------------------------------------------
        # 2. Pick the red cube
        rx, ry, rz = red["pos"]
        pick_z = rz + 0.02                     # 2 cm above centre
        try:
            robot.move_to(rx, ry, pick_z, speed=0.07)
        except Exception:
            robot.log("Pick move failed")
            return False

        robot.grasp()
        robot.move_by(dz=0.01, speed=0.07)    # seat the cube
        if robot.holding() != "red_cube":
            robot.log("Grasp verification failed")
            if overall_attempt == MAX_OVERALL - 1:
                return False
            continue  # restart whole sequence

        # --------------------------------------------------------------
        # 3. Lift clear of the red cube
        robot.move_by(dz=0.02, speed=0.07)

        # --------------------------------------------------------------
        # 4. Move above the blue cube for placement
        bx, by, bz = blue["pos"]
        place_z = bz + 0.015 + 0.015           # centre height of red on blue

        # up to two tries for the placement move (offset check may abort)
        for placement_try in range(2):
            try:
                robot.move_to(bx, by, place_z, speed=0.07)
            except Exception:
                robot.log("Place move failed")
                if overall_attempt == MAX_OVERALL - 1:
                    return False
                break  # abort this overall attempt

            # xy offset from the target is small enough.
            obs2 = robot.observe()
            red2 = obs2.get("objects", {}).get("red_cube")
            if red2 is None:
                robot.log("Red disappeared during placement")
                if overall_attempt == MAX_OVERALL - 1:
                    return False
                break

            dy = red2["pos"][1] - by
            xy_offset = math.hypot(dx, dy)
