coder notes
- Task: grasp red_cube and hold it ≥5 cm above the table. Perception is noisy (~1 cm pos, several deg yaw) and every observe() is a fresh sample, so I average 6 readings per estimate (→ ~4 mm std) and average yaw in 4·angle space to respect the cube's 90° symmetry wrap at ±π/4.
- Grip force/mass/friction vary a lot, so grasping can fail or the cube can slip during the lift: verify with robot.holding() after grasp and after lift, and retry the whole sense→approach→grasp cycle up to 4 times (re-sensing each attempt in case the cube was pushed).
- Approach plan: open fingers, move to (x, y, 0.08) aligned to cube yaw, descend slowly (0.05 m/s) to tool z=0.015 (pads around cube middle, fingertips 2.5 mm above table, safely under the 30 N limit), grasp, then lift straight up to tool z=0.10 so the cube bottom sits ~8.5 cm up with margin over the 5 cm bar.
- All moves that could be out of reach are wrapped in `except Exception` (MotionError isn't importable), with a home() fallback; never hard-code positions — everything comes from observe().

import math

TARGET = "red_cube"
GRASP_Z = 0.015   # tool-point height: finger pads wrap the cube's middle
LIFT_Z = 0.10     # after lift, cube centre ~0.10 m -> bottom ~0.085 m > 5 cm
MAX_ATTEMPTS = 4


def _sense(robot, name, n=6):
    """Average n noisy observations of one object -> ((x, y, z), yaw) or None."""
    sx = sy = sz = 0.0
    s4 = c4 = 0.0
    k = 0
    for _ in range(n):
        obs = robot.observe()
        o = obs["objects"].get(name)
        if o is None:
            continue
        sx += o["pos"][0]
        sy += o["pos"][1]
        sz += o["pos"][2]
        # cube yaw lives in [-pi/4, pi/4): average in 4*angle space to avoid wrap error
        s4 += math.sin(4.0 * o["yaw"])
        c4 += math.cos(4.0 * o["yaw"])
        k += 1
    if k == 0:
        return None
    return (sx / k, sy / k, sz / k), math.atan2(s4, c4) / 4.0


def _try_grasp(robot):
    est = _sense(robot, TARGET)
    if est is None:
        robot.log("target not visible")
        return False
    (x, y, _z), yaw = est
    robot.log("grasp target at", round(x, 4), round(y, 4), "yaw", round(yaw, 3))
    robot.release()
    try:
        robot.move_to(x, y, 0.08, yaw=yaw)
        robot.move_to(x, y, GRASP_Z, yaw=yaw, speed=0.05)
    except Exception as e:
        robot.log("approach failed:", e)
        try:
            robot.home()
        except Exception:
            pass
        return False
    robot.grasp()
    return robot.holding() == TARGET


def _lift_and_check(robot):
    try:
        robot.move_by(dz=LIFT_Z - GRASP_Z, speed=0.08)
    except Exception as e:
        robot.log("lift failed:", e)
        return False
    robot.wait(0.3)
