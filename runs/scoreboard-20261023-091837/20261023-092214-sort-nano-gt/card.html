+

+            robot.move_to(approach_x, approach_y, approach_z,
+                          yaw=None, speed=0.07)
+        except Exception:  # MotionError or any other unexpected exception

+            continue
+

+        # 3b. Lower to the estimated grasp height and close the gripper
+        # ------------------------------------------------------------------
+        grasp_z = 0.028  # fingertip level ≈ centre of a 3 cm cube

+            robot.log("move_to grasp height failed, aborting this object")

+        try:
+            robot.move_to(tray_x, tray_y, release_z,

+        # ------------------------------------------------------------------
